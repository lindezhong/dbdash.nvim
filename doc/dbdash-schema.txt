  # JSON Schema (draft-07, written in YAML) for dbdash.nvim dashboard configs.
  #
  # Editors with yaml-language-server pick it up from a modeline at the top of a config:
  #   # yaml-language-server: $schema=./dbdash.schema.yaml
  #
  # The sidebar (:DbDash) does not list *.schema.yaml files (default sidebar.exclude).
  #
  # Defaults below marked "setup" can be changed for all configs with
  # require("dbdash").setup({ defaults = { ... } }); see :help dbdash-setup.
  #
  # Result format (:help dbdash-result) - referred to by query.sql (lua://, http(s)://) and
  # query.transform below. Every query goes: query -> query.transform (if any) -> a check
  # that the result is rows. SQL databases, Redis, MongoDB and Cassandra return rows; a
  # lua:// script, an HTTP response (decoded JSON) and a transform must return one of:
  #   { columns = { "a", "b" }, rows = { { 1, "x" } } }        rows by position
  #   { columns = { "a", "b" }, rows = { { a = 1, b = "x" } } } rows by name
  #   { { a = 1, b = "x" }, ... }                               records (columns sorted by
  #                                                             name, or a columns list)
  # In JSON: [{"a": 1}] or {"columns": ["a"], "rows": [[1]]}. nil / null is NULL; column
  # types follow the first non-NULL value; query.max_rows applies. Anything else (an
  # object such as {"data": [...]}, a number, text) is an error that suggests a transform.
  $schema: "http://json-schema.org/draft-07/schema#"
  $id: "dbdash.schema.yaml"
  title: dbdash.nvim dashboard
  description: >-
    One dashboard: either a single chart (database / query / show) or several panels
    (panels / layout). Top-level bare fields (interval, args, layout) are dashboard-wide; the
    top-level database, query and show are the panels' defaults (without panels: the only
    panel). See :help dbdash-config.
  type: object
  if: { required: [panels] }
  then:
    # With panels the top-level query and show are defaults: fields describing one chart
    # belong in each panel.
    properties:
      query:
        properties: { sql: false, transform: false, args: false }
      show:
        properties:
          { type: false, title: false, description: false, span: false, column_list: false, x_label: false, y_label: false }
        patternProperties: { "_columns?$": false }
  else:
    required: [database] # query too, except for http(s):// (a GET of the URL)
    not: { required: [layout] } # layout only applies with panels
  dependencies:
    show_countdown: [interval] # it counts down to the global refresh
  properties:

                                            *dbdash-schema-database.url* >yaml
        url:
          type: string
          description: |-
            Connection URL: <scheme>://[user[:password]@]host[:port][/database][?param=value&...]
            User names and passwords are percent-decoded (write @ as %40, : as %3A).
            ${NAME} is replaced by the environment variable NAME ($${ for a literal ${).
            Without a password in the URL: password_cmd, then
            stdpath("data")/dbdash/passwords.json (see :help dbdash-passwords).

            SQLite      sqlite:///absolute/path.db                (or sqlite3://)
                        sqlite://relative/to/this/config.db
                        sqlite:///:memory:
            PostgreSQL  postgres://user:pass@host:5432/db        (or postgresql://)
                        auth: cleartext, md5, SCRAM-SHA-256; ?options=-c%20search_path%3Dapp
            MySQL       mysql://user:pass@host:3306/db           (or mariadb://)
                        auth: mysql_native_password, caching_sha2_password
            Redis       redis://[user]:pass@host:6379/0          (rediss:// = TLS)
            MongoDB     mongodb://user:pass@host:27017/db?authSource=admin
                        mongodb://user:pass@h1:27017,h2:27017/db?readPreference=secondaryPreferred
                        mongodb+srv://user:pass@cluster0.example.net/db   (SRV + TXT lookup,
                        defaults to tls=verify-full; ?dns=host:port picks the name server)
                        ?authMechanism=SCRAM-SHA-256|SCRAM-SHA-1
            Cassandra   cassandra://user:pass@host:9042/keyspace?consistency=LOCAL_QUORUM
                        (or cql://)
            SQL Server  mssql://user:pass@host:1433/db            (or sqlserver://)
                        with TLS: TLS negotiated inside TDS; &encrypt=strict for TDS 8
            Lua         lua://   query.sql is Lua code returning rows (:help dbdash-lua)
            HTTP        https://user:pass@api.example.com/v1?key=value   (or http://)
                        query.sql describes the request (:help dbdash-http); https
                        defaults to tls=verify-full; other ?parameters go to the server

            Common parameters:
              tls=disable|require|pin|verify-ca|verify-full   (default disable)
              fingerprint=sha256:AB:CD:...                    (required for tls=pin)
              ca=path/to/ca.pem                              (verify-*: trusted CAs instead of
                                                              the system store; relative to
                                                              this config file)
              tls_host=name                                  (name to verify instead of host)
              connect_timeout=10s   timeout=30s              (defaults: setup)
              application_name=dbdash.nvim
          examples:
            - "sqlite:///:memory:"
            - "sqlite://data/app.db"
            - "postgres://app:secret@db.internal:5432/app?tls=verify-full"
            - "mysql://report:secret@127.0.0.1:3306/shop?tls=require"
            - "redis://:secret@cache:6379/0"
            - "mongodb+srv://reader:secret@cluster0.example.net/shop"
            - "cassandra://cassandra:cassandra@127.0.0.1:9042/metrics"
            - "mssql://sa:Secret1@sqlhost:1433/master?tls=pin&fingerprint=sha256:AB:..."
            - "lua://"
            - "https://api.example.com/v1?api_key=${API_KEY}"

                                               *dbdash-schema-query.sql* >yaml
        sql:
          type: string
          description: |-
            Query template (Jinja2, see :help dbdash-templates): arguments are
            {{ args.<name> }}, metadata {{ meta.<name> }} (:help dbdash-meta, e.g.
            run records with ids, times and results: meta.run (meta.run.kind: "refresh" or
            "callback"; meta.run.depth: 0 in a refresh, n in a callback), meta.runs,
            meta.prev, meta.prev_ok (the last successful refresh, its rows in
            meta.prev_ok.result): :help dbdash-runs), and
            {% set args.<name> = value %} sets an argument (:help dbdash-assign-args). Lua
            runs in {% lua %} ... {% endlua %} blocks and in modules imported with
            {% import 'x.lua' as h %} (:help dbdash-template-lua), with args, meta, store and
            the dbdash.* helpers (dbdash.run(args): this query again with other arguments);
            a config with Lua is trusted once, with the files it imports (:help dbdash-lua).
            Its language depends on the database:
              SQL databases  SQL (T-SQL for SQL Server, CQL for Cassandra)
              Redis          one command, e.g. HGETALL "user:1001"
              MongoDB        a JSON document:
                             {"collection": "orders", "operation": "find",
                              "filter": {...}, "projection": {...}, "sort": {...},
                              "limit": 100, "flatten": 1}
                             operation: find | aggregate (pipeline) | count |
                             distinct (field) | command (command)
              lua://         Lua code returning rows (see "Result format" at the top
                             of this file, :help dbdash-result), or a value that
                             query.transform turns into rows; it sees args, meta,
                             store and dbdash.system/query/sleep/run (:help
                             dbdash-lua)
              http(s)://     a JSON request description (empty: GET the URL):
                             {"method": "POST", "path": "search", "query": {...},
                              "headers": {...}, "cookies": "a=1; b=2" | "cookies.txt",
                              "body": {...} | "text", "form": {...}, "redirects": 5}
                             JSON responses are decoded, other bodies stay text; the
                             result must be rows ("Result format", :help
                             dbdash-result): a JSON list of objects or {"columns",
                             "rows"} shows directly, anything else needs
                             query.transform (:help dbdash-http)
<
                                         *dbdash-schema-query.transform* >yaml
        transform:
          type: string
          description: |-
            Lua code that reshapes the result of any database before it is shown (after the
            same template as query.sql - args, meta, {% lua %}, .lua imports - without
            autoescape; :help dbdash-transform). Every query goes:
            query -> transform -> rows check. It gets
              rs        SQL and other databases: { columns (names), types, rows (nil for
                        NULL) }; lua://: the script's value; http(s)://: the decoded body
              response  http(s)://: { status, reason, headers, body, url }
              args      the arguments, meta the metadata, store the panel's table kept while
                        the dashboard runs, and the dbdash.* helpers (dbdash.run(args)
                        runs this panel's query again with other arguments)
            and returns rows (see "Result format" at the top of this file, :help
            dbdash-result). Configs with Lua ask for trust first (:help dbdash-lua).
